cmake_minimum_required(VERSION 3.20)
project(native_crash_probe C)
set(CMAKE_C_STANDARD 11)
add_library(diagnostic_original SHARED shim.c)
add_library(diagnostic_guard SHARED shim.c engine_collision_gjk_guard.c)
add_library(diagnostic_fixed SHARED shim.c engine_collision_gjk_fixed.c)
target_compile_definitions(diagnostic_guard PRIVATE mjc_ccd=checked_ccd mjc_ccdSize=checked_ccdSize)
target_compile_definitions(diagnostic_fixed PRIVATE mjc_ccd=fixed_ccd mjc_ccdSize=fixed_ccdSize)
foreach(target diagnostic_original diagnostic_guard diagnostic_fixed)
  target_compile_definitions(${target} PRIVATE MJ_STATIC)
  target_compile_options(${target} PRIVATE /O2 /arch:AVX2)
  target_include_directories(${target} PRIVATE upstream/include upstream/src diagnostic_include)
  target_link_libraries(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/mujoco_import.lib")
endforeach()
